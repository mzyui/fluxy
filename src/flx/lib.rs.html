3//! Fast proxy scraper and validator.
4//!
5//! The [`Flx`] builder mirrors the CLI defaults: scrape the built-in
6//! providers or load candidates from files, validate them against online
7//! judges, then filter, sort, and export the survivors.
8//!
9//! # Examples
10//!
11//! ```no_run
12//! use flx::{Anonymity, Flx, Protocol};
13//!
14//! # async fn example() -> anyhow::Result<()> {
15//! let proxies = Flx::fetch()
16//!     .types([Protocol::Http(Anonymity::Elite)])
17//!     .limit(20)
18//!     .collect()
19//!     .await?;
20//! # Ok(())
21//! # }
22//! ```
23//!
24//! See [`Flx`] for the full builder contract, `examples/` for runnable
25//! workflows, and the README for CLI usage.
26//!
27//! ## Features
28//!
29//! - `log` (default): emit `flx::*` records via the `log` crate.
30//! - `progress_bar` (default): CLI progress rendering.
31//! - `clap` (default): enables the `flx` binary.
32//! - `serve` (optional): rotating proxy endpoint (`flx serve`);
33//!   off by default while experimental.
34//!
35//! ## Minimum Supported Rust Version
36//!
37//! Requires a recent stable toolchain (tested on 1.89+). No `rust-version`
38//! floor is declared in `Cargo.toml`; MSRV bumps are treated as a minor
39//! breaking change and noted in release notes.
40//!
41//! ## License
42//!
43//! Licensed under the MIT license (`LICENSE`).

114/// Writes flx records to stderr with tty-gated colors.
115#[cfg(feature = "log")]
116struct FlxLogger;
117
118#[cfg(feature = "log")]
119const LOG_MODULE_ROOT: &str = "flx";
120
121#[cfg(feature = "log")]
122fn log_module_allowed(target: &str) -> bool {
123    match target.strip_prefix(LOG_MODULE_ROOT) {
124        // Matches Rust module paths split by `::`.
125        Some(rest) => rest.is_empty() || rest.starts_with("::"),
126        None => false,
127    }
128}
129
130#[cfg(feature = "log")]
131fn log_prefix_color(level: log::Level) -> &'static str {
132    match level {
133        log::Level::Error => "\x1b[31m",
134        log::Level::Warn => "\x1b[33m",
135        log::Level::Info => "\x1b[34m",
136        log::Level::Debug => "\x1b[36m",
137        log::Level::Trace => "\x1b[35m",
138    }
139}
140
141#[cfg(feature = "log")]
142impl FlxLogger {
143    fn color_enabled() -> bool {
144        use std::io::IsTerminal as _;
145        match std::env::var_os("TERM") {
146            None => return false,
147            Some(term) => {
148                if term == "dumb" {
149                    return false;
150                }
151            }
152        }
153        if std::env::var_os("NO_COLOR").is_some() {
154            return false;
155        }
156        std::io::stderr().is_terminal()
157    }
158}
159
160#[cfg(feature = "log")]
161impl log::Log for FlxLogger {
162    fn enabled(&self, metadata: &log::Metadata) -> bool {
163        metadata.level() <= log::max_level() && log_module_allowed(metadata.target())
164    }
165
166    fn log(&self, record: &log::Record) {
167        if !self.enabled(record.metadata()) {
168            return;
169        }
170        let mut stderr = std::io::stderr().lock();
171        if Self::color_enabled() {
172            // Keeps prefix bytes identical to the previous logger.
173            let _ = write!(
174                stderr,
175                "\x1b[0m{}{}: {} ",
176                log_prefix_color(record.level()),
177                record.target(),
178                record.level()
179            );
180            let _ = write!(stderr, "\x1b[0m");
181        } else {
182            let _ = write!(stderr, "{}: {} ", record.target(), record.level());
183        }
184        let _ = writeln!(stderr, "{}", record.args());
185    }
186
187    fn flush(&self) {
188        let _ = std::io::stderr().flush();
189    }
190}

192/// Reads proxies from files with per-line protocol pinning.
193///
194/// Bare `ip:port` lines inherit all requested protocols; scheme-prefixed
195/// lines (`socks5://…`) pin their own type. `-` means stdin.
196/// See [`load_proxy_files`] for the file-loading helper.
197///
198/// # Examples
199///
200/// ```no_run
201/// use flx::ProxySource;
202///
203/// # fn example() -> anyhow::Result<()> {
204/// let source = ProxySource::from_reader(std::io::Cursor::new("1.2.3.4:8080\n"))?;
205/// let proxies: Vec<_> = source.collect();
206/// # Ok(())
207/// # }
208/// ```
209pub struct ProxySource {
210    lines: Lines<Box<dyn BufRead + Send>>,
211    default_proxy_types: Arc<[Protocol]>,
212}
213
214/// Defines fallback protocols for bare ip:port lines.
215static FILE_DEFAULT_PROTOCOLS: LazyLock<Arc<[Protocol]>> = LazyLock::new(|| {
216    Arc::from([
217        Protocol::Http(Anonymity::Unknown),
218        Protocol::Https(Anonymity::Unknown),
219        Protocol::Socks4,
220        Protocol::Socks5,
221    ])
222});
223
224/// Formats args into buf, borrowing when it fits.
225pub(crate) fn write_to_buffer<'a>(
226    buf: &'a mut [u8],
227    args: std::fmt::Arguments<'_>,
228) -> Cow<'a, str> {
229    let mut writer = Cursor::new(buf);
230    match writer.write_fmt(args) {
231        Ok(()) => {
232            let len = writer.position() as usize;
233            Cow::Borrowed(std::str::from_utf8(&writer.into_inner()[..len]).expect("ASCII"))
234        }
235        Err(_) => Cow::Owned(args.to_string()),
236    }
237}

239impl ProxySource {
240    /// Builds a fetcher-backed source from `config` without blocking.
241    ///
242    /// # Errors
243    ///
244    /// Returns an error when the provider set cannot be assembled.
245    pub async fn from_fetcher(config: FetcherConfig) -> anyhow::Result<ProxyFetcher> {
246        ProxyFetcher::gather(config).await
247    }
248
249    /// Opens `filepath` for lazy line-by-line parsing.
250    ///
251    /// # Errors
252    ///
253    /// Returns an I/O error when the file cannot be opened.
254    pub fn from_file(filepath: PathBuf) -> anyhow::Result<Self> {
255        let file = anyhow::Context::with_context(File::open(&filepath), || {
256            format!("failed to open proxy file {}", filepath.display())
257        })?;
258        Self::from_reader(BufReader::new(file))
259    }
260
261    /// Wraps any buffered reader as a proxy source.
262    ///
263    /// # Errors
264    ///
265    /// Currently infallible; returns `Ok` for API symmetry with [`ProxySource::from_file`].
266    pub fn from_reader<R: BufRead + Send + 'static>(reader: R) -> anyhow::Result<Self> {
267        let lines = (Box::new(reader) as Box<dyn BufRead + Send>).lines();
268
269        let default_proxy_types = Arc::clone(&FILE_DEFAULT_PROTOCOLS);
270
271        Ok(Self {
272            lines,
273            default_proxy_types,
274        })
275    }
276
277    /// Reads candidates from stdin without blocking.
278    ///
279    /// # Errors
280    ///
281    /// Currently infallible; returns `Ok` for API symmetry with [`ProxySource::from_file`].
282    pub fn from_stdin() -> anyhow::Result<Self> {
283        Self::from_reader(std::io::BufReader::new(std::io::stdin()))
284    }
285}

287impl Iterator for ProxySource {
288    type Item = Proxy;
289
290    fn next(&mut self) -> Option<Self::Item> {
291        for line in self.lines.by_ref() {
292            let line = match line {
293                Ok(line) => line,
294                Err(error) => {
295                    #[cfg(feature = "log")]
296                    log::warn!("failed to read a line from the proxy file: {error}");
297                    #[cfg(not(feature = "log"))]
298                    let _ = error;
299                    continue;
300                }
301            };
302            let line = line.trim();
303            if line.is_empty() {
304                continue;
305            }
306            let mut proxy = match line.parse::<Proxy>() {
307                Ok(proxy) => proxy,
308                Err(error) => {
309                    #[cfg(feature = "log")]
310                    log::warn!("skipped unparseable proxy file line '{line}': {error}");
311                    #[cfg(not(feature = "log"))]
312                    let _ = error;
313                    continue;
314                }
315            };
316            // Pins scheme-prefixed lines; bare lines inherit defaults.
317            if proxy.expected_types.is_empty() {
318                proxy.expected_types = Arc::clone(&self.default_proxy_types);
319            }
320            return Some(proxy);
321        }
322        None
323    }
324}

331    fn proxy_source(content: &str) -> (ProxySource, std::path::PathBuf) {
332        let path = std::env::temp_dir().join(format!(
333            "flx_proxy_source_test_{}_{}.txt",
334            std::process::id(),
335            std::time::SystemTime::now()
336                .duration_since(std::time::UNIX_EPOCH)
337                .unwrap()
338                .as_nanos()
339        ));
340        std::fs::write(&path, content).unwrap();
341        let source = ProxySource::from_file(path.clone()).unwrap();
342        (source, path)
343    }

345    #[test]
346    fn bare_lines_inherit_file_default_protocols() {
347        let (source, path) = proxy_source("1.2.3.4:8080\n5.6.7.8:3128\n\n1.2.3.4:80:Country\n");
348        let proxies: Vec<Proxy> = source.collect();
349        let _ = std::fs::remove_file(&path);
350
351        assert_eq!(proxies.len(), 3);
352        for proxy in &proxies {
353            assert_eq!(
354                proxy.expected_types, *FILE_DEFAULT_PROTOCOLS,
355                "bare lines must inherit the file default protocol set"
356            );
357        }
358    }
