23/// Fetch-phase transitions reported to consumers.
24///
25/// Primary providers run first; fallback mirrors replay only what primary
26/// missed.
27#[derive(Debug, Clone, Copy, PartialEq, Eq)]
28pub enum FetchStage {
29    /// Primary provider set is being scraped.
30    Primary,
31    /// Fallback mirrors are being scraped.
32    Fallback,
33    /// All fetch phases finished.
34    Done,
35}
36
37// Always close with Done, even on early return.
38pub(crate) struct StageReporter {
39    pub(crate) tx: mpsc::Sender<FetchStage>,
40}
41
42impl StageReporter {
43    pub(crate) fn send(&self, stage: FetchStage) {
44        let _ = self.tx.try_send(stage);
45    }
46}
47
48impl Drop for StageReporter {
49    fn drop(&mut self) {
50        let _ = self.tx.try_send(FetchStage::Done);
51    }
52}

68pub(crate) fn spawn_phase(
69    tasks: Vec<(Arc<Source>, Arc<dyn ProxyProvider + Send + Sync>)>,
70    ctx: &PhaseContext,
71) -> JoinSet<()> {
72    let mut handles = JoinSet::new();
73    for (source, provider) in tasks {
74        let job = FetchJob { provider, source };
75        let ctx = ctx.clone();
76        handles.spawn(async move {
77            let url = job.source.url.to_string();
78            if let Err(e) = do_work(job, ctx).await {
79                #[cfg(feature = "log")]
80                log::error!("{}: {:#}", url, e);
81                let _ = (url, e);
82            }
83        });
84    }
85    handles
86}

97/// Serializes network requests to the same host.
98pub(crate) struct Throttle {
99    /// Project next request slot to space concurrent callers apart.
100    next_slot: Mutex<HashMap<String, time::Instant>>,
101}
102
103impl Throttle {
104    pub(crate) fn new() -> Self {
105        Self {
106            next_slot: Mutex::new(HashMap::new()),
107        }
108    }
109
110    pub(crate) async fn wait(&self, host: &str, delay: Duration) {
111        let available_at = {
112            let mut next_slot = self.next_slot.lock().unwrap_or_else(|e| e.into_inner());
113            let now = time::Instant::now();
114            // Reuse key allocation to avoid per-call String alloc.
115            match next_slot.get_mut(host) {
116                Some(previous) => {
117                    let avail_at = (*previous).max(now);
118                    *previous = avail_at + delay;
119                    avail_at
120                }
121                None => {
122                    next_slot.insert(host.to_owned(), now + delay);
123                    now
124                }
125            }
126        };
127        let remaining = available_at.saturating_duration_since(time::Instant::now());
128        if !remaining.is_zero() {
129            time::sleep(remaining).await;
130        }
131    }
132}

134pub(crate) fn source_host(source: &Source) -> String {
135    source
136        .url
137        .host()
138        .map(str::to_owned)
139        .unwrap_or_else(|| source.url.to_string())
140}
141
142async fn throttle_wait(settings: &FetchSettings, source: &Source) {
143    if let Some(delay) = settings.fetch_delay {
144        settings.throttle.wait(&source_host(source), delay).await;
145    }
146}

148pub(crate) async fn do_work(job: FetchJob, ctx: PhaseContext) -> anyhow::Result<()> {
149    let FetchJob { provider, source } = job;
150    if *ctx.stop_rx.borrow() {
151        return Ok(());
152    }
153    let url = source.url.to_string();
154    let expected_types = Arc::clone(&source.default_types);
155
156    let cached = match ctx.settings.fetch_cache.as_ref() {
157        Some(fetch_cache) => fetch_cache.load_rows(&url).await,
158        None => None,
159    };
160
161    let rows: Vec<ParsedProxy> = match cached {
162        Some(rows) => rows,
163        None => {
164            if ctx.settings.offline {
165                #[cfg(feature = "log")]
166                log::warn!("offline: no cached rows for {url}; skipping");
167                return Ok(());
168            }
169            let _permit = ctx
170                .sem
171                .acquire()
172                .await
173                .context("fetcher semaphore closed during shutdown")?;
174            throttle_wait(&ctx.settings, &source).await;
175            let body = provider
176                .fetch(Arc::clone(&ctx.client), &url, source.timeout)
177                .await
178                .with_context(|| format!("failed to fetch proxy list from {}", source.url))?;
179            let mode = source.mode.clone();
180            let rows = tokio::task::spawn_blocking(move || parse_all(&mode, body.as_ref()))
181                .await
182                .context("provider parser task failed")??;
183            if let Some(fetch_cache) = ctx.settings.fetch_cache.as_ref() {
184                fetch_cache.store_rows(&url, &rows).await;
185            }
186            rows
187        }
188    };
189
190    if *ctx.stop_rx.borrow() {
191        return Ok(());
192    }
193
194    for (ip, port, protocol) in rows {
195        let proxy = match protocol {
196            Some(protocol) => Proxy::with_expected_types(ip, port, Arc::from([protocol])),
197            None => Proxy::with_expected_types(ip, port, Arc::clone(&expected_types)),
198        };
199        if ctx.tx.send(proxy).await.is_err() {
200            break;
201        }
202    }
203    Ok(())
204}
